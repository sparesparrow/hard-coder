package io.mcp.client.transport

import io.modelcontextprotocol.kotlin.sdk.types.*
import kotlinx.coroutines.flow.Flow
import org.slf4j.Logger

interface Transport {
    suspend fun connect()
    suspend fun disconnect()
    suspend fun listTools(): List<Tool>
    suspend fun callTool(name: String, arguments: Map<String, Any>? = null): CallToolResult
    fun onProgress(): Flow<Progress>
}

data class Progress(
    val current: Int,
    val total: Int
)

abstract class BaseTransport(protected val logger: Logger) : Transport {
    protected var connected = false

    protected fun checkConnection() {
        if (!connected) {
            throw IllegalStateException("Transport not connected")
        }
    }

    protected fun logError(error: Throwable, context: String) {
        logger.error("$context: ${error.message}", error)
    }
}

    private lateinit var transport: StdioClientTransport

        if (connected) return

        try {
            client = Client(
                clientInfo = Implementation(
                    name = "kotlin-mcp-client",
                    version = "1.0.0"
                )
            )

            transport = StdioClientTransport(
                command = command,
                args = args
            )

            connected = true
